"""Append-only resume after the documented review-controller import setup error."""
from pathlib import Path
import sys
sys.path.insert(0,str(Path(__file__).resolve().parent))
import resume as x
import controls as c
import extra
extra.x=x;extra.c=c
# Bind this supplemental controller in every new receipt, not just its imports.
original_run=x.run
me=Path(__file__).resolve()
def guarded_run(label,args,extra=None,outputs=(),dependencies=()):
    return original_run(label,args,extra,outputs,(*dependencies,me))
x.run=guarded_run
need=x.need;R=x.R
src='''use vstd::prelude::*;use std::num::NonZero;
#[verus_verify(infer_derive(PartialEq,PartialOrd))]
#[derive(Copy,Clone,Hash,PartialEq,Eq,PartialOrd)] pub struct BorTag(NonZero<u64>);
'''
local=x.json.loads((R/'partial-only-local/receipt.json').read_text())
need(local['exit']==0 and local['guards'] and local['verification']['verified']==6,'retained complete local proof')
prior=(R/'final-bool-prior/wrapper.vir').read_bytes();rows={}
for goals in [False,True]:
    for route in ['explicit','cargo','both','equal']:
        label='partial-request-'+('on' if goals else 'off')+'-'+route
        d=R/label;d.mkdir(exist_ok=True);cargo=d/'libwrapper.vir';explicit=cargo if route=='equal' else d/'selected.vir'
        selected=[explicit] if route=='explicit' else [cargo] if route=='cargo' else list(dict.fromkeys([explicit,cargo]))
        if (d/'receipt.json').exists():
            j=x.json.loads((d/'receipt.json').read_text());need(j['exit']!=0 and j['guards'],'retained refusal')
        else:
            for p in selected:p.write_bytes(prior)
            c.expect(label,src,ok=False,goals=goals,cargo=route!='explicit',export=explicit if route!='cargo' else None)
        need('local-only' in (d/'stderr').read_text() and all(not p.exists() for p in selected),'nohelper all-selected refusal')
        rows[label]=dict(prior_sha256=x.hashlib.sha256(prior).hexdigest(),removed=list(map(str,selected)))
x.write(R/'partial-only-invalidation.json',rows)
extra.proposals()
